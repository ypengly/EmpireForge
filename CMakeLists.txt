cmake_minimum_required(VERSION 3.21)
project(EmpireForge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
include(FetchContent)
find_package(SFML 3 QUIET COMPONENTS Graphics Window System Audio)
if(NOT SFML_FOUND)
    FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 3.0.0 GIT_SHALLOW ON)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(SFML)
endif()
find_package(SQLite3 QUIET)
if(NOT SQLite3_FOUND)
    FetchContent_Declare(sqlite3 URL https://www.sqlite.org/2024/sqlite-amalgamation-3450100.zip)
    FetchContent_MakeAvailable(sqlite3)
    add_library(sqlite3 STATIC ${sqlite3_SOURCE_DIR}/sqlite3.c)
    target_include_directories(sqlite3 PUBLIC ${sqlite3_SOURCE_DIR})
    set(SQLite3_LIBRARIES sqlite3)
else()
    set(SQLite3_LIBRARIES SQLite::SQLite3)
endif()
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW ON)
FetchContent_MakeAvailable(json)
file(GLOB_RECURSE EF_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_executable(EmpireForge ${EF_SOURCES})
target_include_directories(EmpireForge PRIVATE ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(EmpireForge PRIVATE SFML::Graphics SFML::Window SFML::System SFML::Audio ${SQLite3_LIBRARIES} nlohmann_json::nlohmann_json)
add_custom_command(TARGET EmpireForge POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/data $<TARGET_FILE_DIR:EmpireForge>/data)
add_custom_command(TARGET EmpireForge POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:EmpireForge>/assets)
enable_testing()
add_subdirectory(tests)
